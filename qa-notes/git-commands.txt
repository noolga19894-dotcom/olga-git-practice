Git commands
pwd — показати поточну папку

ls — показати файли та папки

ls -a — показати також приховані файли

cd — перейти в папку

cd .. — перейти на рівень вище

cd ~ — перейти в домашню папку

cd - — повернутися до попередньої папки

mkdir — створити папку

touch — створити файл

echo — записати текст

> — записати текст у файл із перезаписом

>> — додати текст у кінець файлу


git status — показати стан репозиторію

git branch — показати гілки

git switch — перейти на іншу гілку

git switch -c — створити нову гілку та одразу перейти на неї

git add — додати зміни до області підготовки

git add * — додати всі зміни з поточної папки

git commit -m — створити коміт із повідомленням

git push — відправити коміти на GitHub

git push -u origin — відправити нову гілку на GitHub і зв’язати її з віддаленою

git clone — клонувати репозиторій з GitHub
